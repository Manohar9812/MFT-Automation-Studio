			<participant name="BPMetaDataInfoService"/>
			<output message="BPMetaDataServiceTypeInputMessage">

			<input message="inmsg">

							<assign to="Variables" from="*"/>

						<participant name="MailboxExtractBegin"/>
						<output message="MailboxExtractBeginServiceTypeInputMessage">

							<assign to="Variables/Message/MessagePath" from="string(MailboxPath)" append="true"></assign>
							<assign to="PrimaryDocument" from="PrimaryDocument/@SCIObjectID"></assign>
							<assign to="Variables/Message/MessageSize" from="string(MessageSize)" append="true"></assign>
						</input>
					</operation>
					<operation name="Get Document Information Service">
						<participant name="GetDocumentInfoService"/>
						<output message="GetDocumentInfoServiceTypeInputMessage">
							<assign to="DocumentName" from="/ProcessData/Variables/Message/MessageName/text()"/>
							<assign to="updateMetaDataOnly">true</assign>
							<assign to="." from="*"/>
						</output>
						<input message="inmsg">
						</input>
					</operation>
					<operation name="Put File">
						<participant name="InlineInvokeBusinessProcessService"/>
						<output message="InvokeBusinessProcessServiceTypeInputMessage">
							<assign to="WFD_NAME">SCE_SFTPPut</assign>
							<assign to="." from="*"/>
						</output>
						<input message="inmsg">
							<assign to="." from="*"/>
						</input>
					</operation>
					<operation name="DeleteFile">
						<participant name="MailboxDelete"/>
						<output message="DeleteRequest">
							<assign to="MessageId" from="string(RoutingRequest/RoutingRequest/MessageId[number(/ProcessData/counters/positionCurrent)])"/>
							<assign to="DaysOld">0</assign>

							<assign to="DeleteResults" from="*"/>
						</input>
					</operation>

						<output message="XSLTServiceTypeInputMessage">
							<assign to="xslt_name">SCE_SFTP_Put_Multiple_Success_MailBody</assign>
							<assign to="xml_input_from">ProcData</assign>
							<assign to="input_pd_xpath">/ProcessData</assign>
							<assign to="xml_input_validation">NO</assign>
							<assign to="." from="*"/>
						</output>
						<input message="inmsg">
							<assign to="ReadXSLTResults" from="*"/>
						</input>
					</operation>

							<assign to="mail-mime-subject-encoding">UTF-8</assign>

							<assign to="mail-mime-body">true</assign>
							<assign to="." from="*"/>
						</output>
						<input message="inmsg">
							<assign to="." from="*"/>
						</input>
					</operation>
					<operation name="SMTP Send Email">
						<participant name="SMTP_SEND_ADAPTER"/>
						<output message="Xout">
							<assign to="." from="*"/>
							<assign to="xport-smtp-mailfrom">scemftsys@sce.com</assign>
							<assign to="xport-smtp-mailhost" from="//mailhost/text()"/>
							<assign to="b2b-raw-message">true</assign>
							<assign to="xport-smtp-mailport">25</assign>

							<assign to="." from="*"/>

					</operation>

						<participant name="ReleaseService"/>
						<output message="ReleaseServiceTypeInputMessage">
							<assign to="TARGET">ReadXSLTResults</assign>

						</output>
						<input message="inmsg">
							<assign to="." from="*"/>
						</input>
					</operation>
					<operation name="ReleasePD">
						<participant name="ReleaseService"/>
						<output message="ReleaseServiceTypeInputMessage">
							<assign to="TARGET">PrimaryDocument | DocumentName | DocumentId | MessageName | ArchiveInfo | message_to_child | TargetFileName | MessageSize | Variables</assign>
							<assign to="." from="*"/>
						</output>
						<input message="inmsg">
							<assign to="." from="*"/>
						</input>
					</operation>

							<spawn name="Spawn" ref="SCE_Generic_EMailOnError"/>

								<participant name="BPExceptionService"/>
								<output message="BPExceptionServiceTypeInputMessage">
									<assign to="exceptionCode">ERROR</assign>
									<assign to="." from="*"/>
								</output>
								<input message="inmsg">
									<assign to="." from="*"/>
								</input>
							</operation>
						</sequence>
					</onFault>
				</sequence>
			</process>
